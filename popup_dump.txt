     1	/** @odoo-module */
     2	
     3	import { Component } from "@odoo/owl";
     4	import { Dialog } from "@web/core/dialog/dialog";
     5	
     6	export const SPICE_LEVELS = [
     7	    { value: "1", label: "Anti Pedas" },
     8	    { value: "2", label: "Sedang" },
     9	    { value: "3", label: "Pedas" },
    10	    { value: "4", label: "Extra Pedas" },
    11	    { value: "5", label: "Level Neraka" },
    12	];
    13	
    14	// Popup dibangun di atas komponen Dialog generik dari @web/core (bukan
    15	// service "popup" lama yang sudah dihapus dari POS sejak refactor OWL).
    16	// Dipanggil lewat this.dialog.add(SpiceLevelPopup, { productName, getPayload }).
    17	export class SpiceLevelPopup extends Component {
    18	    static template = "geprekyukss_pos.SpiceLevelPopup";
    19	    static components = { Dialog };
    20	    static props = {
    21	        productName: String,
    22	        close: Function,
    23	        getPayload: Function,
    24	    };
    25	
    26	    setup() {
    27	        this.levels = SPICE_LEVELS;
    28	    }
    29	
    30	    selectLevel(value) {
    31	        this.props.getPayload(value);
    32	        this.props.close();
    33	    }
    34	
    35	    cancel() {
    36	        this.props.getPayload(null);
    37	        this.props.close();
    38	    }
    39	}
